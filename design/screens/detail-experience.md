  const modal = document.getElementById('experienceModal');
  const backdrop = document.getElementById('modalBackdrop');
  const container = document.getElementById('modalContainer');
  const closeBtn = document.getElementById('modalCloseBtn');
  const dismissFooterBtn = document.getElementById('modalDismissFooterBtn');

  const modalBadge = document.getElementById('modalBadge');
  const modalActiveStatus = document.getElementById('modalActiveStatus');
  const modalDuration = document.getElementById('modalDuration');
  const modalRoleTitle = document.getElementById('modalRoleTitle');
  const modalCompany = document.getElementById('modalCompany');
  const modalLocation = document.getElementById('modalLocation');
  const modalSummary = document.getElementById('modalSummary');
  const modalAchievements = document.getElementById('modalAchievements');
  const modalTechStack = document.getElementById('modalTechStack');

  // Open modal with specific role data
  function openExperienceModal(id) {
    const data = experienceData[id];
    if (!data) return;

    // Populate modal contents
    modalBadge.textContent = data.initials;
    if (data.isDarkBadge) {
      modalBadge.className = "w-14 h-14 shrink-0 rounded-2xl flex items-center justify-center text-lg font-extrabold shadow-sm bg-on-background text-primary-container";
    } else {
      modalBadge.className = "w-14 h-14 shrink-0 rounded-2xl flex items-center justify-center text-lg font-extrabold shadow-sm bg-surface-container-high text-on-surface";
    }

    if (data.active) {
      modalActiveStatus.classList.remove('hidden');
    } else {
      modalActiveStatus.classList.add('hidden');
    }

    modalDuration.textContent = data.period;
    modalRoleTitle.textContent = data.title;
    modalCompany.textContent = data.company;
    modalLocation.innerHTML = `<span class="material-symbols-outlined text-[14px]">location_on</span> ${data.location}`;
    modalSummary.textContent = data.summary;

    // Achievements list
    modalAchievements.innerHTML = data.achievements.map(item => `
      <li class="flex gap-2.5 items-start">
        <span class="inline-block h-2 w-2 rounded-full bg-primary-container shrink-0 mt-1.5 shadow-[0_0_0_1px_#161c26]"></span>
        <span class="leading-relaxed">${item}</span>
      </li>
    `).join('');

    // Tech stack badges
    modalTechStack.innerHTML = data.technologies.map(tech => `
      <span class="inline-flex items-center px-2.5 py-1 font-semibold rounded-md border bg-surface-container-high font-label-sm text-label-sm text-on-surface border-outline-variant/40">
        ${tech}
      </span>
    `).join('');

    // Show modal
    modal.classList.remove('hidden');
    document.body.style.overflow = 'hidden';

    // Animate in
    requestAnimationFrame(() => {
      backdrop.classList.remove('opacity-0');
      backdrop.classList.add('opacity-100');
      container.classList.remove('scale-95', 'opacity-0');
      container.classList.add('scale-100', 'opacity-100');
    });
  }

  function closeExperienceModal() {
    backdrop.classList.remove('opacity-100');
    backdrop.classList.add('opacity-0');
    container.classList.remove('scale-100', 'opacity-100');
    container.classList.add('scale-95', 'opacity-0');

    setTimeout(() => {
      modal.classList.add('hidden');
      document.body.style.overflow = '';
    }, 200);
  }

  // Attach event listeners
  document.querySelectorAll('[data-experience-id]').forEach(el => {
    el.addEventListener('click', (e) => {
      const id = el.getAttribute('data-experience-id');
      if (id) openExperienceModal(id);
    });

    el.addEventListener('keydown', (e) => {
      if (e.key === 'Enter' || e.key === ' ') {
        e.preventDefault();
        const id = el.getAttribute('data-experience-id');
        if (id) openExperienceModal(id);
      }
    });
  });

  closeBtn.addEventListener('click', closeExperienceModal);
  dismissFooterBtn.addEventListener('click', closeExperienceModal);
  backdrop.addEventListener('click', closeExperienceModal);

  // Close on ESC
  document.addEventListener('keydown', (e) => {
    if (e.key === 'Escape' && !modal.classList.contains('hidden')) {
      closeExperienceModal();
    }
  });
</script>
